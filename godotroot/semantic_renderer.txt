extends Node3D

# ─────────────────────────────────────────────
#  SemanticRenderer.gd
#  3D world, 2D trixel atlas tiles on XZ plane.
#  Atlas loading ported from trixel_atlas_smoke.gd.
#  TODAY: entity spawning + autotiled terrain.
#  NOT TODAY: events, AP, HUD, snapshot sync.
# ─────────────────────────────────────────────

const ASSET_ROOT  := "res://engain/tests/trixel/assets"
const SCENE_ID    := "03_Fist_contact"
const TILE_SIZE   := 1.0       # world units per tile
const TERRAIN_Y   := 0.0       # ground plane
const TILE_PX     := 16        # atlas tile pixel size

const TERRAIN_DRAW_ORDER := [
	"deep_water","shallow_water","shoreline","sand",
	"grass","forest_edge","pier","rock","cliff",
]
const FALLBACK_TERRAIN := "grass"

@export var entity_scene: PackedScene

var _atlases: Dictionary = {}     # terrain → {texture, rects}
var _terrain_grid: Array  = []    # 2D array of terrain strings
var _grid_w: int = 0
var _grid_h: int = 0


# ── Boot ──────────────────────────────────────
func _ready() -> void:
	print("[SemanticRenderer] Ready")
	if entity_scene == null:
		entity_scene = load("res://entities/TrixelEntity3D.tscn")

	_load_all_atlases()

	if not _validate_scene_client():
		return
	SceneClient.scene_loaded.connect(_on_scene_loaded)
	SceneClient.scene_error.connect(_on_scene_error)
	SceneClient.load_scene(SCENE_ID)


# ── Atlas loading (ported from trixel_atlas_smoke) ─
func _load_all_atlases() -> void:
	_atlases.clear()
	for terrain in TERRAIN_DRAW_ORDER:
		var dir      := "%s/%s" % [ASSET_ROOT, terrain]
		var man_path := "%s/manifest.json" % dir
		var meta_path:= "%s/atlas_meta.json" % dir
		var atlas_path:= "%s/atlas.png" % dir

		if not FileAccess.file_exists(man_path) \
		or not FileAccess.file_exists(meta_path):
			push_warning("[SemanticRenderer] Missing atlas files for: " + terrain)
			continue

		var manifest   = JSON.parse_string(FileAccess.get_file_as_string(man_path))
		var atlas_meta = JSON.parse_string(FileAccess.get_file_as_string(meta_path))
		var tex: Texture2D = load(atlas_path)

		if typeof(manifest) != TYPE_DICTIONARY \
		or typeof(atlas_meta) != TYPE_DICTIONARY \
		or tex == null:
			push_warning("[SemanticRenderer] Bad atlas data for: " + terrain)
			continue

		var tile_order: Array = atlas_meta.get("tile_order", [])
		var columns: int      = int(atlas_meta.get("columns", 4))
		var tw: int           = int(atlas_meta.get("tile_width",  TILE_PX))
		var th: int           = int(atlas_meta.get("tile_height", TILE_PX))
		var rects: Dictionary = {}

		for i in tile_order.size():
			var role: String = str(tile_order[i])
			var col: int = i % columns
			var row: int = i / columns
			rects[role] = Rect2i(col * tw, row * th, tw, th)

		_atlases[terrain] = { "texture": tex, "rects": rects }

	print("[SemanticRenderer] Loaded %d terrain atlases" % _atlases.size())


# ── Scene response ─────────────────────────────
func _on_scene_loaded(scene_doc: Dictionary) -> void:
	var data: Dictionary = scene_doc.get("data", scene_doc)

	if data.has("metadata"):
		print("[SemanticRenderer] scene_id: ", data["metadata"].get("scene_id", "?"))

	_clear_scene()

	# build terrain grid then render
	if data.has("terrain_chunks"):
		_build_grid_from_commands(data["terrain_chunks"])
	elif data.has("terrain"):
		_build_grid_from_raw(data["terrain"])
	else:
		_build_fallback_grid(16, 16)

	_render_terrain()

	# spawn entities on top
	for cmd in data.get("spawn_commands", []):
		if cmd.get("type", "spawn_entity") == "spawn_entity":
			_spawn_entity(cmd)


func _on_scene_error(msg: String) -> void:
	push_error("[SemanticRenderer] Scene load failed: " + msg)


# ── Terrain grid builders ──────────────────────
func _build_grid_from_commands(chunks: Array) -> void:
	# find extents
	var max_x := 0
	var max_z := 0
	for c in chunks:
		max_x = max(max_x, int(c.get("x", 0)))
		max_z = max(max_z, int(c.get("z", 0)))
	_init_grid(max_x + 1, max_z + 1, FALLBACK_TERRAIN)
	for c in chunks:
		_terrain_grid[int(c.get("z", 0))][int(c.get("x", 0))] = c.get("type", FALLBACK_TERRAIN)


func _build_grid_from_raw(raw) -> void:
	if raw is Array and raw.size() > 0 and raw[0] is Array:
		_grid_h = raw.size()
		_grid_w = raw[0].size()
		_terrain_grid = raw.duplicate(true)
	else:
		_build_fallback_grid(16, 16)


func _build_fallback_grid(w: int, h: int) -> void:
	print("[SemanticRenderer] Using fallback terrain grid %dx%d" % [w, h])
	_init_grid(w, h, FALLBACK_TERRAIN)


func _init_grid(w: int, h: int, fill: String) -> void:
	_grid_w = w
	_grid_h = h
	_terrain_grid.clear()
	for _z in range(h):
		var row := []
		row.resize(w)
		row.fill(fill)
		_terrain_grid.append(row)


# ── Terrain renderer — one mesh per atlas ─────
# Collects all tiles into per-terrain SurfaceTool batches.
# Result: 1 MeshInstance3D + 1 draw call per terrain type,
# not one node per tile.
func _render_terrain() -> void:
	if _terrain_grid.is_empty():
		return
	print("[SemanticRenderer] Baking terrain %dx%d" % [_grid_w, _grid_h])

	# st_map: terrain → SurfaceTool
	var st_map: Dictionary = {}

	for z in _grid_h:
		for x in _grid_w:
			var terrain: String = str(_terrain_grid[z][x])
			if not _atlases.has(terrain):
				terrain = FALLBACK_TERRAIN
			if not _atlases.has(terrain):
				continue
			var role := _resolve_role(x, z, terrain)
			if not st_map.has(terrain):
				var st := SurfaceTool.new()
				st.begin(Mesh.PRIMITIVE_TRIANGLES)
				st_map[terrain] = st
			_add_tile_to_surface(st_map[terrain], x, z, terrain, role)

	# commit one MeshInstance3D per terrain type
	for terrain in st_map:
		var st: SurfaceTool = st_map[terrain]
		st.generate_normals()
		var mesh_inst := MeshInstance3D.new()
		mesh_inst.name = "Terrain_" + terrain
		mesh_inst.mesh = st.commit()
		var mat := StandardMaterial3D.new()
		mat.albedo_texture = _atlases[terrain]["texture"]
		mat.texture_filter = BaseMaterial3D.TEXTURE_FILTER_NEAREST
		mesh_inst.material_override = mat
		add_child(mesh_inst)

	print("[SemanticRenderer] Terrain baked — %d mesh(es)" % st_map.size())


func _add_tile_to_surface(st: SurfaceTool, tx: int, tz: int,
		terrain: String, role: String) -> void:
	var atlas: Dictionary = _atlases[terrain]
	var rects: Dictionary = atlas["rects"]
	var tex: Texture2D    = atlas["texture"]
	var tex_w := float(tex.get_width())
	var tex_h := float(tex.get_height())

	var region: Rect2i = rects.get(role, rects.get("center", Rect2i(0, 0, TILE_PX, TILE_PX)))

	# UV corners for this atlas region
	var u0 := region.position.x / tex_w
	var v0 := region.position.y / tex_h
	var u1 := (region.position.x + region.size.x) / tex_w
	var v1 := (region.position.y + region.size.y) / tex_h

	# flat XZ quad at TERRAIN_Y
	var x0 := tx * TILE_SIZE
	var x1 := x0 + TILE_SIZE
	var z0 := tz * TILE_SIZE
	var z1 := z0 + TILE_SIZE
	var y  := TERRAIN_Y

	# two triangles, normal up
	var nrm := Vector3.UP

	st.set_uv(Vector2(u0, v0)); st.set_normal(nrm); st.add_vertex(Vector3(x0, y, z0))
	st.set_uv(Vector2(u1, v0)); st.set_normal(nrm); st.add_vertex(Vector3(x1, y, z0))
	st.set_uv(Vector2(u1, v1)); st.set_normal(nrm); st.add_vertex(Vector3(x1, y, z1))

	st.set_uv(Vector2(u0, v0)); st.set_normal(nrm); st.add_vertex(Vector3(x0, y, z0))
	st.set_uv(Vector2(u1, v1)); st.set_normal(nrm); st.add_vertex(Vector3(x1, y, z1))
	st.set_uv(Vector2(u0, v1)); st.set_normal(nrm); st.add_vertex(Vector3(x0, y, z1))


# ── Autotile role resolver (from trixel_atlas_smoke) ─
func _terrain_at(x: int, z: int) -> String:
	if z < 0 or z >= _terrain_grid.size(): return ""
	var row: Array = _terrain_grid[z]
	if x < 0 or x >= row.size(): return ""
	return str(row[x])


func _resolve_role(x: int, z: int, terrain: String) -> String:
	var n := _terrain_at(x, z - 1) == terrain
	var s := _terrain_at(x, z + 1) == terrain
	var e := _terrain_at(x + 1, z) == terrain
	var w := _terrain_at(x - 1, z) == terrain
	var count := int(n) + int(s) + int(e) + int(w)

	match count:
		0: return "single"
		4: return "center"
		3:
			if not n: return "edge_n"
			if not s: return "edge_s"
			if not e: return "edge_e"
			return "edge_w"
		2:
			if s and e:  return "corner_nw"
			if s and w:  return "corner_ne"
			if n and e:  return "corner_sw"
			if n and w:  return "corner_se"
			if n and s:  return "path_straight_v"
			return "path_straight_h"
		1:
			if n: return "path_end_s"
			if s: return "path_end_n"
			if e: return "path_end_w"
			return "path_end_e"
	return "center"


# ── Entity spawning ────────────────────────────
func _spawn_entity(cmd: Dictionary) -> void:
	if entity_scene == null:
		push_error("[SemanticRenderer] entity_scene is null")
		return
	var ent = entity_scene.instantiate()
	ent.name = cmd.get("id", "entity")
	if cmd.has("position"):
		var p = cmd["position"]
		ent.position = Vector3(p[0], p[1], p[2])
	add_child(ent)
	if ent.has_method("apply_render_plan"):
		ent.apply_render_plan(cmd)
	elif ent.has_method("apply_metadata"):
		ent.apply_metadata(cmd)
	print("[SemanticRenderer] Spawned: ", ent.name)


# ── Cleanup ────────────────────────────────────
func _clear_scene() -> void:
	_terrain_grid.clear()
	_grid_w = 0
	_grid_h = 0
	for c in get_children():
		c.queue_free()


# ── Guard ─────────────────────────────────────
func _validate_scene_client() -> bool:
	if not Engine.has_singleton("SceneClient") \
	and not get_node_or_null("/root/SceneClient"):
		push_error("[SemanticRenderer] SceneClient autoload not found")
		return false
	return true
