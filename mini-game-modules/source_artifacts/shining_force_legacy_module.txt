**Battle Transition System:**
```gdscript
enum EnemyType {
    RANDOM_ENCOUNTER,    # Invisible, probability-based
    ROAMING_PATROL,      # Visible, moves in patterns
    STATIONARY_GUARD,    # Blocks paths/treasures
    AGGRESSIVE_CHASER,   # Pursues player actively
    AMBUSH_PREDATOR     # Hidden until triggered
}

func trigger_battle(enemy_data, encounter_context):
    # Store overworld state
    var battle_setup = {
        "enemy_formation": enemy_data.formation,
        "terrain_type": get_current_terrain(),
        "encounter_type": enemy_data.type,
        "player_advantage": calculate_initiative_bonus(),
        "environmental_effects": get_local_hazards()
    }
    
    # Transition to tactical grid
    SceneManager.transition_to_tactical_battle(battle_setup)
```
