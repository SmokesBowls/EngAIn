# Event Sourcing Layer for Dream State Integrity
extends Node
class_name DreamEventStore

# Event sourcing for dream state reconstruction
var event_log = []
var event_checkpoints = {}
var max_events_per_checkpoint = 100

# Event types for dream system
enum EventType {
    DREAM_ENTERED,
    PHYSICS_MODIFIED,
    ENTROPY_CHANGED,
    SNAPSHOT_CREATED,
    CORRUPTION_DETECTED,
    ANCHOR_APPLIED,
    REALITY_SHIFTED
}

func record_event(type: int, data: Dictionary, system_id: String = "dream"):
    var event = {
        "id": generate_event_id(),
        "type": type,
        "data": data,
        "timestamp": OS.get_ticks_msec(),
        "system_id": system_id,
        "dream_depth": DreamStateManager.dream_depth,
        "entropy_state": RealityEngine.timeline_entropy
    }
    
    event_log.append(event)
    
    # Create checkpoint every N events
    if event_log.size() % max_events_per_checkpoint == 0:
        create_event_checkpoint()
    
    # Emit for real-time debugging
    EventBus.emit_signal("dream_event_recorded", event)

func create_event_checkpoint():
    var checkpoint_id = "checkpoint_" + str(OS.get_unix_time())
    var checkpoint = {
        "events_count": event_log.size(),
        "system_snapshot": capture_system_snapshot(),
        "timestamp": OS.get_ticks_msec(),
        "integrity_hash": calculate_integrity_hash()
    }
    
    event_checkpoints[checkpoint_id] = checkpoint

func replay_events_from_checkpoint(checkpoint_id: String):
    """Reconstruct dream state by replaying events from checkpoint"""
    if not event_checkpoints.has(checkpoint_id):
        push_error("Checkpoint not found: " + checkpoint_id)
        return false
    
    var checkpoint = event_checkpoints[checkpoint_id]
    var start_index = checkpoint.events_count
    
    # Restore to checkpoint state
    restore_system_snapshot(checkpoint.system_snapshot)
    
    # Replay subsequent events
    for i in range(start_index, event_log.size()):
        var event = event_log[i]
        replay_single_event(event)
    
    return true

func replay_single_event(event: Dictionary):
    """Replay a single event to reconstruct system state"""
    match event.type:
        EventType.DREAM_ENTERED:
            # Reconstruct dream entry without full initialization
            DreamStateManager.dream_depth = event.data.depth
            DreamStateManager.current_state = DreamStateManager.DreamState.DREAMING
            
        EventType.PHYSICS_MODIFIED:
            # Apply physics modification
            var physics_data = event.data
            PhysicsStack.apply_physics_mod(physics_data.name, physics_data.params)
            
        EventType.ENTROPY_CHANGED:
            # Restore entropy state
            RealityEngine.timeline_entropy = event.data.new_entropy
            DreamStateManager.dream_entropy = event.data.dream_entropy
            
        EventType.CORRUPTION_DETECTED:
            # Re-apply corruption handling if still relevant
            if validate_corruption_relevance(event):
                DreamStateManager.handle_dream_corruption(event.data.corruption_type)

func validate_corruption_relevance(event: Dictionary) -> bool:
    """Check if a corruption event is still relevant to current state"""
    var current_entropy = RealityEngine.timeline_entropy
    var event_entropy = event.entropy_state
    
    # Only replay if entropy is similar (corruption might still exist)
    return abs(current_entropy - event_entropy) < 10.0

func capture_system_snapshot():
    """Lightweight snapshot for event checkpoints"""
    return {
        "physics_stack": PhysicsStack.physics_mod_stack.duplicate(true),
        "dream_depth": DreamStateManager.dream_depth,
        "dream_state": DreamStateManager.current_state,
        "entropy_levels": {
            "timeline": RealityEngine.timeline_entropy,
            "dream": DreamStateManager.dream_entropy
        },
        "active_anchors": Global.locked_anchor
    }

func restore_system_snapshot(snapshot: Dictionary):
    """Restore system from lightweight snapshot"""
    # Restore physics stack
    PhysicsStack.physics_mod_stack = snapshot.physics_stack.duplicate(true)
    PhysicsStack.apply_current_stack()
    
    # Restore dream state
    DreamStateManager.dream_depth = snapshot.dream_depth
    DreamStateManager.current_state = snapshot.dream_state
    
    # Restore entropy
    RealityEngine.timeline_entropy = snapshot.entropy_levels.timeline
    DreamStateManager.dream_entropy = snapshot.entropy_levels.dream
    
    # Restore anchors
    Global.locked_anchor = snapshot.active_anchors

func calculate_integrity_hash() -> String:
    """Calculate hash for corruption detection"""
    var hash_input = ""
    hash_input += str(DreamStateManager.dream_depth)
    hash_input += str(RealityEngine.timeline_entropy)
    hash_input += str(PhysicsStack.physics_mod_stack.size())
    
    return hash_input.sha256_text()

func validate_system_integrity() -> bool:
    """Validate current system state against expected integrity"""
    var current_hash = calculate_integrity_hash()
    
    # Check against most recent checkpoint
    var latest_checkpoint = get_latest_checkpoint()
    if not latest_checkpoint:
        return true  # No checkpoint to compare against
    
    # Allow for some entropy drift
    var expected_components = latest_checkpoint.integrity_hash.split("")
    var current_components = current_hash.split("")
    
    var differences = 0
    for i in range(min(expected_components.size(), current_components.size())):
        if expected_components[i] != current_components[i]:
            differences += 1
    
    # Accept up to 20% hash difference (accounts for normal entropy changes)
    return (differences / float(current_hash.length())) < 0.2

func get_latest_checkpoint():
    """Get the most recent checkpoint"""
    var latest_time = 0
    var latest_checkpoint = null
    
    for checkpoint_id in event_checkpoints.keys():
        var checkpoint = event_checkpoints[checkpoint_id]
        if checkpoint.timestamp > latest_time:
            latest_time = checkpoint.timestamp
            latest_checkpoint = checkpoint
    
    return latest_checkpoint

func emergency_state_reconstruction():
    """Last resort: reconstruct state from event log"""
    push_warning("Emergency state reconstruction initiated")
    
    # Find last valid checkpoint
    var valid_checkpoint = find_last_valid_checkpoint()
    if not valid_checkpoint:
        # No valid checkpoint - full reset required
        DreamStateManager.emergency_reality_reset()
        return false
    
    # Replay from checkpoint
    var success = replay_events_from_checkpoint(valid_checkpoint)
    if not success:
        DreamStateManager.emergency_reality_reset()
        return false
    
    # Verify integrity after reconstruction
    if not validate_system_integrity():
        push_error("State reconstruction failed integrity check")
        DreamStateManager.emergency_reality_reset()
        return false
    
    return true

func find_last_valid_checkpoint():
    """Find the most recent checkpoint that passes validation"""
    var checkpoints = event_checkpoints.keys()
    checkpoints.sort()
    
    for i in range(checkpoints.size() - 1, -1, -1):
        var checkpoint_id = checkpoints[i]
        var checkpoint = event_checkpoints[checkpoint_id]
        
        # Validate checkpoint integrity
        if validate_checkpoint_integrity(checkpoint):
            return checkpoint_id
    
    return null

func validate_checkpoint_integrity(checkpoint: Dictionary) -> bool:
    """Validate that a checkpoint is not corrupted"""
    # Check required fields
    var required_fields = ["events_count", "system_snapshot", "timestamp", "integrity_hash"]
    for field in required_fields:
        if not checkpoint.has(field):
            return false
    
    # Check timestamp reasonableness
    var age = OS.get_ticks_msec() - checkpoint.timestamp
    if age < 0 or age > 3600000:  # More than 1 hour old
        return false
    
    # Validate system snapshot structure
    var snapshot = checkpoint.system_snapshot
    if not snapshot.has("physics_stack") or not snapshot.has("dream_depth"):
        return false
    
    return true

func cleanup_old_events():
    """Periodically clean up old events to prevent memory bloat"""
    var max_age = 1800000  # 30 minutes in milliseconds
    var current_time = OS.get_ticks_msec()
    
    # Remove old events
    var filtered_events = []
    for event in event_log:
        if current_time - event.timestamp < max_age:
            filtered_events.append(event)
    
    event_log = filtered_events
    
    # Remove old checkpoints
    var valid_checkpoints = {}
    for checkpoint_id in event_checkpoints.keys():
        var checkpoint = event_checkpoints[checkpoint_id]
        if current_time - checkpoint.timestamp < max_age:
            valid_checkpoints[checkpoint_id] = checkpoint
    
    event_checkpoints = valid_checkpoints

func generate_event_id() -> String:
    """Generate unique event ID"""
    return str(OS.get_ticks_msec()) + "_" + str(randi())

# Integration with existing systems
func _ready():
    # Connect to dream state manager
    DreamStateManager.connect("dream_state_changed", self, "_on_dream_state_changed")
    DreamStateManager.connect("dream_corrupted", self, "_on_dream_corrupted")
    
    # Connect to reality engine
    RealityEngine.connect("reality_shift", self, "_on_reality_shift")
    
    # Setup cleanup timer
    var cleanup_timer = Timer.new()
    cleanup_timer.wait_time = 300.0  # Clean up every 5 minutes
    cleanup_timer.timeout.connect(self, "cleanup_old_events")
    cleanup_timer.autostart = true
    add_child(cleanup_timer)

func _on_dream_state_changed(old_state, new_state):
    record_event(EventType.DREAM_ENTERED, {
        "old_state": old_state,
        "new_state": new_state,
        "depth": DreamStateManager.dream_depth
    })

func _on_dream_corrupted(corruption_type, recovery_options):
    record_event(EventType.CORRUPTION_DETECTED, {
        "corruption_type": corruption_type,
        "recovery_options": recovery_options,
        "system_state": capture_system_snapshot()
    })

func _on_reality_shift(intensity, affected_nodes):
    record_event(EventType.REALITY_SHIFTED, {
        "intensity": intensity,
        "affected_nodes": affected_nodes,
        "entropy_before": RealityEngine.timeline_entropy
    })
