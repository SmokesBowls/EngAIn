# Enhanced Dream-State System with robust edge case handling
extends Node
class_name DreamStateManager

# Core dream state tracking
enum DreamState { AWAKE, ENTERING, DREAMING, EXITING, CORRUPTED }
var current_state = DreamState.AWAKE
var dream_depth = 0  # Track nested dreams
var max_dream_depth = 3

# Entropy management
var dream_entropy = 0.0
var entropy_decay_rate = 0.5
var entropy_corruption_threshold = 85.0

# State preservation
var reality_snapshots = {}
var snapshot_compression = true
var max_snapshots = 5

# Performance monitoring
var dream_performance_metrics = {
    "frame_drops": 0,
    "memory_spikes": 0,
    "corruption_events": 0
}

signal dream_state_changed(old_state, new_state)
signal dream_corrupted(corruption_type, recovery_options)
signal nested_dream_limit_reached()

func _ready():
    # Connect to RealityEngine for entropy synchronization
    if RealityEngine:
        RealityEngine.connect("reality_shift", self, "_on_reality_shift")
        RealityEngine.connect("paradox_detected", self, "_on_paradox_detected")

func attempt_dream_entry(dream_type = "standard"):
    # Comprehensive pre-flight checks
    var validation_result = validate_dream_entry()
    if not validation_result.success:
        handle_dream_entry_failure(validation_result.reason)
        return false
    
    # State transition with rollback capability
    var old_state = current_state
    current_state = DreamState.ENTERING
    
    if not execute_dream_entry(dream_type):
        # Rollback on failure
        current_state = old_state
        return false
    
    emit_signal("dream_state_changed", old_state, current_state)
    return true

func validate_dream_entry():
    var result = {"success": true, "reason": ""}
    
    # Check system state conflicts
    if Global.locked_anchor != "":
        result.success = false
        result.reason = "reality_anchored"
        return result
    
    # Check dream depth limits
    if dream_depth >= max_dream_depth:
        result.success = false
        result.reason = "max_depth_reached"
        emit_signal("nested_dream_limit_reached")
        return result
    
    # Check for active paradoxes
    if RealityEngine.timeline_entropy > entropy_corruption_threshold:
        result.success = false
        result.reason = "timeline_unstable"
        return result
    
    # Memory availability check
    var available_memory = OS.get_static_memory_usage_by_type()
    if available_memory < 50 * 1024 * 1024:  # 50MB threshold
        result.success = false
        result.reason = "insufficient_memory"
        return result
    
    # UI system availability
    if UI.current_priority >= 2:  # Stellar broadcast or paradox resolution active
        result.success = false
        result.reason = "ui_priority_conflict"
        return result
    
    return result

func execute_dream_entry(dream_type):
    # Create compressed reality snapshot
    var snapshot_id = "dream_entry_" + str(OS.get_unix_time())
    if not create_quantum_snapshot(snapshot_id):
        push_error("Failed to create reality snapshot")
        return false
    
    # Initialize dream environment with error handling
    if not initialize_dream_physics():
        restore_snapshot(snapshot_id)
        return false
    
    if not load_dream_scene(dream_type):
        restore_snapshot(snapshot_id)
        cleanup_dream_physics()
        return false
    
    # Finalize state transition
    dream_depth += 1
    current_state = DreamState.DREAMING
    start_entropy_management()
    
    return true

func create_quantum_snapshot(snapshot_id):
    var snapshot_data = {
        "timestamp": OS.get_unix_time(),
        "reality_entropy": RealityEngine.timeline_entropy,
        "player_state": serialize_player_state(),
        "world_state": serialize_world_state(),
        "ui_state": UI.get_state_snapshot(),
        "dream_depth": dream_depth
    }
    
    # Compress if enabled and size is significant
    if snapshot_compression and get_snapshot_size(snapshot_data) > 1024:
        snapshot_data = compress_snapshot(snapshot_data)
    
    # Manage snapshot storage limits
    if reality_snapshots.size() >= max_snapshots:
        var oldest_key = reality_snapshots.keys().min()
        reality_snapshots.erase(oldest_key)
    
    reality_snapshots[snapshot_id] = snapshot_data
    return true

func serialize_player_state():
    return {
        "position": Global.player.global_position,
        "velocity": Global.player.velocity if Global.player.has_method("get_velocity") else Vector2.ZERO,
        "health": Global.player_health,
        "inventory": Global.inventory.duplicate(true),
        "abilities": Global.unlocked_abilities.duplicate(),
        "quantum_state": RealityEngine.get_quantum_state("player").duplicate() if RealityEngine.quantum_states.has("player") else null
    }

func serialize_world_state():
    var world_state = {}
    
    # Capture essential world objects
    for node in get_tree().get_nodes_in_group("persistent"):
        if node.has_method("get_save_data"):
            world_state[node.get_path()] = node.get_save_data()
    
    # Environmental state
    world_state["gravity"] = Physics2DServer.area_get_param(get_world_2d().space, Physics2DServer.AREA_PARAM_GRAVITY)
    world_state["time_scale"] = Engine.time_scale
    
    return world_state

func initialize_dream_physics():
    # Store original physics settings
    var original_gravity = Physics2DServer.area_get_param(get_world_2d().space, Physics2DServer.AREA_PARAM_GRAVITY)
    
    # Apply dream physics with validation
    var dream_gravity = -9.8 * (1.0 + dream_entropy * 0.01)
    
    if abs(dream_gravity) > 50.0:  # Sanity check
        push_error("Dream gravity exceeds safe limits: " + str(dream_gravity))
        return false
    
    Physics2DServer.area_set_param(get_world_2d().space, Physics2DServer.AREA_PARAM_GRAVITY, dream_gravity)
    return true

func load_dream_scene(dream_type):
    var scene_path = get_dream_scene_path(dream_type)
    
    # Validate scene exists and is loadable
    if not ResourceLoader.exists(scene_path):
        push_error("Dream scene not found: " + scene_path)
        return false
    
    # Load with error handling
    var scene_resource = load(scene_path)
    if not scene_resource:
        push_error("Failed to load dream scene: " + scene_path)
        return false
    
    var dream_instance = scene_resource.instance()
    if not dream_instance:
        push_error("Failed to instance dream scene")
        return false
    
    # Add to scene tree with validation
    get_tree().current_scene.add_child(dream_instance)
    dream_instance.name = "DreamRealm_" + str(dream_depth)
    
    # Initialize dream-specific systems
    if dream_instance.has_method("initialize_dream_logic"):
        dream_instance.initialize_dream_logic(dream_entropy)
    
    return true

func start_entropy_management():
    # Create entropy decay timer with bounds checking
    var entropy_timer = Timer.new()
    entropy_timer.wait_time = 1.0  # Update every second
    entropy_timer.autostart = true
    entropy_timer.connect("timeout", self, "_on_entropy_tick")
    add_child(entropy_timer)

func _on_entropy_tick():
    # Decay dream entropy over time
    dream_entropy = max(0.0, dream_entropy - entropy_decay_rate)
    
    # Check for corruption events
    if dream_entropy > entropy_corruption_threshold:
        handle_dream_corruption("entropy_overflow")
        return
    
    # Update dream environment based on entropy
    update_dream_environment()

func update_dream_environment():
    var dream_realm = get_node_or_null("DreamRealm_" + str(dream_depth))
    if not dream_realm:
        return
    
    var intensity = clamp(dream_entropy / 100.0, 0.0, 1.0)
    
    # Safe shader parameter updates
    if dream_realm.has_node("DreamShader"):
        var shader_node = dream_realm.get_node("DreamShader")
        if shader_node.material and shader_node.material.shader:
            shader_node.material.set_shader_param("distortion", intensity * 2.0)
    
    # NPC behavior modifications with bounds checking
    for npc in dream_realm.get_nodes_in_group("dream_npcs"):
        if npc.has_method("set_aggression"):
            var aggression = lerp(0.1, 0.9, intensity)
            npc.set_aggression(clamp(aggression, 0.0, 1.0))

func handle_dream_corruption(corruption_type):
    dream_performance_metrics.corruption_events += 1
    
    var recovery_options = []
    
    match corruption_type:
        "entropy_overflow":
            recovery_options = ["force_wake", "entropy_reset", "emergency_save"]
        "memory_leak":
            recovery_options = ["garbage_collect", "force_wake"]
        "infinite_recursion":
            recovery_options = ["force_wake", "stack_reset"]
    
    emit_signal("dream_corrupted", corruption_type, recovery_options)
    
    # Automatic recovery for critical failures
    if corruption_type in ["memory_leak", "infinite_recursion"]:
        force_wake_player()

func force_wake_player():
    current_state = DreamState.EXITING
    
    # Find most recent valid snapshot
    var latest_snapshot_id = get_latest_valid_snapshot()
    if latest_snapshot_id:
        restore_snapshot(latest_snapshot_id)
    else:
        # Emergency fallback
        emergency_reality_reset()
    
    cleanup_dream_state()
    current_state = DreamState.AWAKE

func get_latest_valid_snapshot():
    var valid_snapshots = []
    for snapshot_id in reality_snapshots.keys():
        var snapshot = reality_snapshots[snapshot_id]
        if validate_snapshot_integrity(snapshot):
            valid_snapshots.append(snapshot_id)
    
    return valid_snapshots.max() if valid_snapshots.size() > 0 else null

func validate_snapshot_integrity(snapshot):
    # Check for required fields
    var required_fields = ["timestamp", "reality_entropy", "player_state", "world_state"]
    for field in required_fields:
        if not snapshot.has(field):
            return false
    
    # Validate timestamp is reasonable
    var age = OS.get_unix_time() - snapshot.timestamp
    if age > 3600:  # Older than 1 hour
        return false
    
    return true

func restore_snapshot(snapshot_id):
    if not reality_snapshots.has(snapshot_id):
        push_error("Snapshot not found: " + snapshot_id)
        return false
    
    var snapshot = reality_snapshots[snapshot_id]
    
    # Restore player state
    if snapshot.has("player_state"):
        restore_player_state(snapshot.player_state)
    
    # Restore world state
    if snapshot.has("world_state"):
        restore_world_state(snapshot.world_state)
    
    # Restore reality entropy
    if snapshot.has("reality_entropy"):
        RealityEngine.timeline_entropy = snapshot.reality_entropy
    
    return true

func cleanup_dream_state():
    # Remove dream realm
    var dream_realm = get_node_or_null("DreamRealm_" + str(dream_depth))
    if dream_realm:
        dream_realm.queue_free()
    
    # Reset physics
    cleanup_dream_physics()
    
    # Decrement dream depth
    dream_depth = max(0, dream_depth - 1)
    
    # Reset entropy
    dream_entropy = 0.0

func cleanup_dream_physics():
    # Restore original gravity from world state
    var latest_snapshot = get_latest_valid_snapshot()
    if latest_snapshot and reality_snapshots[latest_snapshot].has("world_state"):
        var world_state = reality_snapshots[latest_snapshot].world_state
        if world_state.has("gravity"):
            Physics2DServer.area_set_param(get_world_2d().space, Physics2DServer.AREA_PARAM_GRAVITY, world_state.gravity)

func emergency_reality_reset():
    push_warning("Emergency reality reset triggered")
    
    # Reset all major systems to safe defaults
    RealityEngine.timeline_entropy = 50.0
    dream_entropy = 0.0
    dream_depth = 0
    current_state = DreamState.AWAKE
    
    # Reset physics to default
    Physics2DServer.area_set_param(get_world_2d().space, Physics2DServer.AREA_PARAM_GRAVITY, 9.8)
    
    # Clear all snapshots
    reality_snapshots.clear()
    
    # Show emergency notification
    if UI.has_method("show_emergency_notice"):
        UI.show_emergency_notice("Reality stabilized. Some progress may be lost.")

# Performance monitoring hooks
func _on_reality_shift(intensity, affected_nodes):
    if current_state == DreamState.DREAMING:
        # Amplify reality shifts in dreams
        dream_entropy += intensity * 5.0
        
        # Monitor for performance issues
        if intensity > 2.0:
            dream_performance_metrics.frame_drops += 1

func _on_paradox_detected(severity, resolution_options):
    if current_state == DreamState.DREAMING:
        # Dreams are more susceptible to paradoxes
        if severity == "critical":
            handle_dream_corruption("paradox_cascade")
        else:
            dream_entropy += 15.0
