from bge import logic

print("[EngAIn] Script loaded!")

controller = logic.getCurrentController()
owner = controller.owner
scene = logic.getCurrentScene()

if not hasattr(logic, 'initialized'):
    print("[EngAIn] === INIT CALLED ===")
    print(f"[EngAIn] Scene: {scene.name}")
    print(f"[EngAIn] Objects in scene: {len(scene.objects)}")
    
    logic.initialized = True
    
    # Spawn 5 cubes in a row
    for i in range(5):
        try:
            cube = scene.addObject("EntityTemplate", "spawnpoint")
            cube.worldPosition.x = i * 2  # Space them 2 units apart
            cube.color = [1, 0, 0, 1]  # Red color
            print(f"[EngAIn] Spawned cube {i} at X={i * 2}")
        except Exception as e:
            print(f"[EngAIn] ERROR spawning cube {i}: {e}")
    
    print("[EngAIn] Init complete!")
